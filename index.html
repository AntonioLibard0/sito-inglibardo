modbus_crc.c
/**
 * Calcola il CRC-16 Modbus (polinomio 0xA001, riflesso)
 * di un frame RTU. Il risultato va trasmesso LSB first.
 */
uint16_t modbus_crc16(const uint8_t *buf, size_t len)
{
    uint16_t crc = 0xFFFF;

    for (size_t i = 0; i < len; i++) {
        crc ^= buf[i];
        for (uint8_t b = 0; b < 8; b++) {
            crc = (crc & 1u) ? (crc >> 1) ^ 0xA001
                              : (crc >> 1);
        }
    }
    return crc;
}
